
dmsd_strip_location = {
	if = {
		limit = { has_variable_list = dmsd_court_applied }

		every_in_list = {
			variable = dmsd_court_applied
			save_temporary_scope_as = dmsd_old_demand
			scope:dmsd_market = { remove_temporary_demand = scope:dmsd_old_demand }
		}

		clear_variable_list = dmsd_court_applied
	}

	if = {
		limit = { has_variable_list = dmsd_diplo_applied }

		every_in_list = {
			variable = dmsd_diplo_applied
			save_temporary_scope_as = dmsd_old_demand
			scope:dmsd_market = { remove_temporary_demand = scope:dmsd_old_demand }
		}

		clear_variable_list = dmsd_diplo_applied
	}

	if = {
		limit = { has_variable_list = dmsd_stab_applied }

		every_in_list = {
			variable = dmsd_stab_applied
			save_temporary_scope_as = dmsd_old_demand
			scope:dmsd_market = { remove_temporary_demand = scope:dmsd_old_demand }
		}

		clear_variable_list = dmsd_stab_applied
	}

	if = {
		limit = { has_variable_list = dmsd_cult_applied }

		every_in_list = {
			variable = dmsd_cult_applied
			save_temporary_scope_as = dmsd_old_demand
			scope:dmsd_market = { remove_temporary_demand = scope:dmsd_old_demand }
		}

		clear_variable_list = dmsd_cult_applied
	}

	if = { limit = { has_variable_map = dmsd_court_map } clear_variable_map = dmsd_court_map }

	if = { limit = { has_variable_map = dmsd_diplo_map } clear_variable_map = dmsd_diplo_map }

	if = { limit = { has_variable_map = dmsd_stab_map } clear_variable_map = dmsd_stab_map }

	if = { limit = { has_variable_map = dmsd_cult_map } clear_variable_map = dmsd_cult_map }
}

dmsd_full_cleanup = {
	dmsd_nuke_all_demands = yes

	every_market_in_world = {
		location = {
			if = { limit = { has_variable_list = dmsd_court_applied } clear_variable_list = dmsd_court_applied }
			if = { limit = { has_variable_list = dmsd_diplo_applied } clear_variable_list = dmsd_diplo_applied }
			if = { limit = { has_variable_list = dmsd_stab_applied } clear_variable_list = dmsd_stab_applied }
			if = { limit = { has_variable_list = dmsd_cult_applied } clear_variable_list = dmsd_cult_applied }
			if = { limit = { has_variable_map = dmsd_court_map } clear_variable_map = dmsd_court_map }
			if = { limit = { has_variable_map = dmsd_diplo_map } clear_variable_map = dmsd_diplo_map }
			if = { limit = { has_variable_map = dmsd_stab_map } clear_variable_map = dmsd_stab_map }
			if = { limit = { has_variable_map = dmsd_cult_map } clear_variable_map = dmsd_cult_map }
		}
	}

	clear_global_variable_list = dmsd_active_locations
	# Clear per-country proximity caches and display variables
	random_country = {
		limit = { is_real_country = yes }
		every_country = {
			if = { limit = { has_variable_list = dmsd_prox_sources } clear_variable_list = dmsd_prox_sources }
			if = { limit = { has_variable_map = dmsd_prox_weights } clear_variable_map = dmsd_prox_weights }
			if = { limit = { has_variable = dmsd_prox_total } remove_variable = dmsd_prox_total }
			if = { limit = { has_variable = dmsd_last_gov_power } remove_variable = dmsd_last_gov_power }
			if = { limit = { has_variable = dmsd_court_slider } remove_variable = dmsd_court_slider }
			if = { limit = { has_variable = dmsd_last_stability } remove_variable = dmsd_last_stability }
			if = { limit = { has_variable = dmsd_stab_slider } remove_variable = dmsd_stab_slider }
			if = { limit = { has_variable = dmsd_cult_raw_gold } remove_variable = dmsd_cult_raw_gold }
			if = { limit = { has_variable = dmsd_disp_court_rel_dict } remove_variable = dmsd_disp_court_rel_dict }
			# Per-source display maps
			if = { limit = { has_variable_map = dmsd_prox_court_dict } clear_variable_map = dmsd_prox_court_dict }
			if = { limit = { has_variable_map = dmsd_prox_court_gold } clear_variable_map = dmsd_prox_court_gold }
			if = { limit = { has_variable_map = dmsd_prox_court_scale } clear_variable_map = dmsd_prox_court_scale }
			if = { limit = { has_variable_map = dmsd_prox_diplo_dict } clear_variable_map = dmsd_prox_diplo_dict }
			if = { limit = { has_variable_map = dmsd_prox_diplo_gold } clear_variable_map = dmsd_prox_diplo_gold }
			if = { limit = { has_variable_map = dmsd_prox_diplo_scale } clear_variable_map = dmsd_prox_diplo_scale }
			if = { limit = { has_variable_map = dmsd_prox_stab_dict } clear_variable_map = dmsd_prox_stab_dict }
			if = { limit = { has_variable_map = dmsd_prox_stab_gold } clear_variable_map = dmsd_prox_stab_gold }
			if = { limit = { has_variable_map = dmsd_prox_stab_scale } clear_variable_map = dmsd_prox_stab_scale }
			if = { limit = { has_variable_map = dmsd_prox_cult_dict } clear_variable_map = dmsd_prox_cult_dict }
			if = { limit = { has_variable_map = dmsd_prox_cult_gold } clear_variable_map = dmsd_prox_cult_gold }
			if = { limit = { has_variable_map = dmsd_prox_cult_scale } clear_variable_map = dmsd_prox_cult_scale }
		}
	}

	if = { limit = { has_global_variable = dmsd_version } remove_global_variable = dmsd_version }

	if = { limit = { has_global_variable = dmsd_swap } remove_global_variable = dmsd_swap }

	if = { limit = { has_global_variable_map = dmsd_court_reg1 } clear_global_variable_map = dmsd_court_reg1 }

	if = { limit = { has_global_variable_map = dmsd_diplo_reg1 } clear_global_variable_map = dmsd_diplo_reg1 }

	if = { limit = { has_global_variable_map = dmsd_stab_reg1 } clear_global_variable_map = dmsd_stab_reg1 }

	if = { limit = { has_global_variable_map = dmsd_demand_swap } clear_global_variable_map = dmsd_demand_swap }

	if = { limit = { has_global_variable_map = dmsd_demand_dict } clear_global_variable_map = dmsd_demand_dict }

	if = { limit = { has_global_variable_map = dmsd_cult_reg1 } clear_global_variable_map = dmsd_cult_reg1 }

	if = { limit = { has_global_variable = dmsd_dict_ref } remove_global_variable = dmsd_dict_ref }

	# Clear weight maps from allocated location dicts
	every_in_global_list = {
		variable = dmsd_all_dicts
		if = { limit = { has_variable_map = dmsd_weights } clear_variable_map = dmsd_weights }
	}

	clear_global_variable_list = dmsd_all_dicts
}
