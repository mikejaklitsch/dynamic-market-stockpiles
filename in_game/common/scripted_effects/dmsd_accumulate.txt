# Phase 1 accumulation: distributes gold across proximity sources.
# Each source gets gold proportional to its proximity strength.
# Court splits 50/50 between regional (per-source subcontinent) and religion.
# Diplo/Stab use regional only (per-source subcontinent).
#
# Expected context: country scope
#   scope:dmsd_rg       = religion group (court only)
#   local_var:dmsd_gold = total gold for this slider
#   dmsd_prox_sources   = cached proximity source locations
#   dmsd_prox_weights   = cached proximity strengths
#   var:dmsd_prox_total = total proximity strength

dmsd_accumulate_court = {
	clear_variable_map = dmsd_prox_court_io
	clear_variable_map = dmsd_prox_court_gold
	clear_variable_map = dmsd_prox_court_scale
	every_in_list = {
		variable = dmsd_prox_sources
		save_temporary_scope_as = dmsd_src
		sub_continent = { save_temporary_scope_as = dmsd_sc }

		set_local_variable = {
			name = dmsd_share
			value = {
				value = local_var:dmsd_gold
				multiply = "prev.variable_map(dmsd_prox_weights|scope:dmsd_src)"
				divide = prev.var:dmsd_prox_total
			}
		}

		# Determine the regional IO for this source's subcontinent
		if = {
			limit = { is_key_in_global_variable_map = { name = dmsd_court_reg1 target = scope:dmsd_sc } }
			set_local_variable = { name = dmsd_disp_d value = "global_variable_map(dmsd_court_reg1|scope:dmsd_sc)" }
		} else = {
			set_local_variable = { name = dmsd_disp_d value = demand:dmsd_court_demand1 }
		}
		set_local_variable = { name = dmsd_disp_io value = "global_variable_map(dmsd_demand_io|local_var:dmsd_disp_d)" }

		# Compute basket cost at this source's market: sum(weight × market_price)
		# The scale = gold / basket_cost gives actual demand units per good
		market = { save_temporary_scope_as = dmsd_src_mkt }
		set_local_variable = { name = dmsd_basket_cost value = 0 }
		set_global_variable = { name = dmsd_io_ref value = local_var:dmsd_disp_io }
		global_var:dmsd_io_ref = {
			every_key_in_variable_map = {
				variable = dmsd_weights
				save_temporary_scope_as = dmsd_good
				scope:dmsd_src_mkt = {
					change_local_variable = {
						name = dmsd_basket_cost
						add = { value = "global_var:dmsd_io_ref.variable_map(dmsd_weights|scope:dmsd_good)" multiply = "market_price(scope:dmsd_good)" }
					}
				}
			}
		}

		# Store IO ref and demand scale (gold / basket_cost) for GUI display
		set_local_variable = { name = dmsd_src_scale value = 0 }
		if = {
			limit = { local_var:dmsd_basket_cost > 0 }
			set_local_variable = { name = dmsd_src_scale value = { value = local_var:dmsd_share divide = local_var:dmsd_basket_cost } }
		}
		prev = {
			add_to_variable_map = { name = dmsd_prox_court_io key = scope:dmsd_src value = local_var:dmsd_disp_io }
			add_to_variable_map = { name = dmsd_prox_court_gold key = scope:dmsd_src value = local_var:dmsd_share }
			add_to_variable_map = { name = dmsd_prox_court_scale key = scope:dmsd_src value = local_var:dmsd_src_scale }
		}

		market.location = {
			add_to_global_variable_list = { name = dmsd_active_locations target = this }

			if = {
				limit = { is_key_in_global_variable_map = { name = dmsd_court_reg1 target = scope:dmsd_sc } }
				set_local_variable = { name = dmsd_map_key value = "global_variable_map(dmsd_court_reg1|scope:dmsd_sc)" }
				if = {
					limit = { is_key_in_global_variable_map = { name = dmsd_court_reg1 target = scope:dmsd_rg } }
					set_local_variable = { name = dmsd_half value = { value = local_var:dmsd_share divide = 2 } }

					# Accumulate regional half
					set_local_variable = { name = dmsd_map_val value = local_var:dmsd_half }
					if = {
						limit = { is_key_in_variable_map = { name = dmsd_court_map target = local_var:dmsd_map_key } }
						change_local_variable = { name = dmsd_map_val add = "variable_map(dmsd_court_map|local_var:dmsd_map_key)" }
					}
					dmsd_set_variable_map = { name = dmsd_court_map key = local_var:dmsd_map_key value = local_var:dmsd_map_val }

					# Accumulate religion half
					set_local_variable = { name = dmsd_map_key value = "global_variable_map(dmsd_court_reg1|scope:dmsd_rg)" }
					set_local_variable = { name = dmsd_map_val value = local_var:dmsd_half }
					if = {
						limit = { is_key_in_variable_map = { name = dmsd_court_map target = local_var:dmsd_map_key } }
						change_local_variable = { name = dmsd_map_val add = "variable_map(dmsd_court_map|local_var:dmsd_map_key)" }
					}
					dmsd_set_variable_map = { name = dmsd_court_map key = local_var:dmsd_map_key value = local_var:dmsd_map_val }
				} else = {
					# Regional only — full share
					set_local_variable = { name = dmsd_map_val value = local_var:dmsd_share }
					if = {
						limit = { is_key_in_variable_map = { name = dmsd_court_map target = local_var:dmsd_map_key } }
						change_local_variable = { name = dmsd_map_val add = "variable_map(dmsd_court_map|local_var:dmsd_map_key)" }
					}
					dmsd_set_variable_map = { name = dmsd_court_map key = local_var:dmsd_map_key value = local_var:dmsd_map_val }
				}
			} else = {
				# Fallback — no regional variant
				set_local_variable = { name = dmsd_map_val value = local_var:dmsd_share }
				if = {
					limit = { is_key_in_variable_map = { name = dmsd_court_map target = demand:dmsd_court_demand1 } }
					change_local_variable = { name = dmsd_map_val add = "variable_map(dmsd_court_map|demand:dmsd_court_demand1)" }
				}
				dmsd_set_variable_map = { name = dmsd_court_map key = demand:dmsd_court_demand1 value = local_var:dmsd_map_val }
			}
		}
	}
}
