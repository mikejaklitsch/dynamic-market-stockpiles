
namespace = dmsd_slider_demand

# ═══════════════════════════════════════════════════════════════════════════════
# SLIDER DEMAND EVENTS
# ═══════════════════════════════════════════════════════════════════════════════
# Phase 1 of the monthly demand cycle. Fires per-country via
# monthly_country_pulse. Each slider computes its gold, applies
# efficiency decay, then distributes across proximity sources into
# market location maps. Phase 2 runs in weather_monthly_pulse after all
# countries have accumulated, reading the populated maps and applying
# market demands: spending measured in one month becomes market demand
# the next month.

# --- Per-slider efficiency decay ---
# Diminishing returns on heavy spending. Each doubling of raw gold past
# the baseline costs 25% efficiency. Per-slider, not country-wide, so a
# large nation spending modestly is unpenalized.
@dmsd_eff_baseline = 250
# 0.415 = log(0.75)/log(0.5), the exponent that yields exactly ×0.75 per doubling.
@dmsd_eff_exponent = 0.415

# ═══════════════════════════════════════════════════════════════════════════════
# PHASE 1: PER-COUNTRY ACCUMULATORS
# ═══════════════════════════════════════════════════════════════════════════════

# --- Event .1: Court spending ---
# Gold = inferred_slider × 0.1 × eco_base / efficiency_divisor
# Court splits 50/50 between regional and religion demand baskets.

dmsd_slider_demand.1 = {
	type = country_event
	title = hidden_event.t
	hidden = yes

	immediate = {
		if = {
			limit = { NOT = { has_variable_list = dmsd_prox_sources } }
			dmsd_rebuild_prox_cache = yes
		}

		religion = { group = { save_temporary_scope_as = dmsd_rel_group } }
		dmsd_infer_court_slider = yes

		# Resolve the religion-group basket dict for GUI tooltip display.
		# The expense tooltip reads dmsd_disp_court_rel_dict via GetVariable;
		# the religion basket is country-level, so it cannot come from the
		# per-source display maps like the regional baskets do.
		if = {
			limit = { is_key_in_global_variable_map = { name = dmsd_court_reg1 target = scope:dmsd_rel_group } }
			set_local_variable = { name = dmsd_disp_demand value = "global_variable_map(dmsd_court_reg1|scope:dmsd_rel_group)" }

			set_variable = {
				name = dmsd_disp_court_rel_dict
				value = "global_variable_map(dmsd_demand_dict|local_var:dmsd_disp_demand)"
			}
		}

		# Court gold calculation.
		# court_maintenance trigger gives raw slider position (0-1) but doesn't
		# account for reduced spending when gov power is capped. We use the
		# inferred effective slider from dmsd_infer_court_slider instead.
		# Engine clamps the efficiency divisor to min 0.5 (max 2× penalty).
		set_local_variable = {
			name = dmsd_gold
			value = {
				value = 0
				if = {
					limit = { has_variable = dmsd_court_slider }
					add = {
						value = var:dmsd_court_slider
						multiply = 0.1
						multiply = country_economical_base

						divide = {
							value = 1
							add = modifier:court_spending_efficiency
							min = 0.5
						}
					}
				}
			}
		}

		# Efficiency decay applied to pre-decay gold; see constants above
		change_local_variable = {
			name = dmsd_gold
			multiply = {
				value = @dmsd_eff_baseline
				divide = { value = local_var:dmsd_gold min = @dmsd_eff_baseline }
				pow = @dmsd_eff_exponent
			}
		}

		dmsd_accumulate_court = yes
	}
}

# --- Event .2: Diplomatic spending ---
# Gold = diplomatic_maintenance × cost × (1 + cost) × eco_base / efficiency
# Vanilla applies diplomatic_spending_cost twice: once as the base rate, once
# as a (1 + cost) multiplier.

dmsd_slider_demand.2 = {
	type = country_event
	title = hidden_event.t
	hidden = yes

	immediate = {
		set_local_variable = {
			name = dmsd_gold
			value = {
				value = diplomatic_maintenance
				multiply = modifier:diplomatic_spending_cost
				multiply = { value = 1 add = modifier:diplomatic_spending_cost }
				multiply = country_economical_base

				divide = {
					value = 1
					add = modifier:diplomatic_upkeep_efficiency
					min = 0.5
				}
			}
		}

		change_local_variable = {
			name = dmsd_gold
			multiply = {
				value = @dmsd_eff_baseline
				divide = { value = local_var:dmsd_gold min = @dmsd_eff_baseline }
				pow = @dmsd_eff_exponent
			}
		}

		dmsd_accumulate_diplo = yes
	}
}

# --- Event .5: Cultural spending ---
# Cultural cost comes from artist salaries. Verified against two saves
# 2026-07-24 (exact match on France with mixed salary modifiers):
#   monthly_per_artist = ARTIST_SALARY_BASE_FACTOR × tax_base ÷ artist_limit
#                        × (1 + artist_salary_modifier + char_modifier)
# Salaries are monthly values, the divisor is the artist limit
# (num_possible_artists), and the country-wide and per-character salary
# modifiers combine additively per artist. Summed over employed artists:
#   total = 0.025 × maint × tax_base / limit × (N × (1 + country_mod) + Σ char_mods)

dmsd_slider_demand.5 = {
	type = country_event
	title = hidden_event.t
	hidden = yes

	immediate = {
		set_local_variable = { name = dmsd_num_artists value = 0 }
		set_local_variable = { name = dmsd_char_mod_sum value = 0 }

		every_artist = {
			change_local_variable = { name = dmsd_num_artists add = 1 }
			change_local_variable = { name = dmsd_char_mod_sum add = modifier:artist_salary_modifier_on_character }
		}

		set_local_variable = { name = dmsd_artist_limit value = modifier:num_possible_artists }

		set_local_variable = {
			name = dmsd_gold
			value = {
				value = local_var:dmsd_num_artists
				multiply = { value = 1 add = modifier:artist_salary_modifier }
				add = local_var:dmsd_char_mod_sum
				multiply = 0.025
				multiply = cultural_maintenance
				multiply = country_tax_base
				divide = { value = local_var:dmsd_artist_limit min = 1 }
			}
		}

		# Stored for tooltip; rebuilding it in a script value requires
		# every_artist iteration.
		set_variable = { name = dmsd_cult_raw_gold value = local_var:dmsd_gold }

		change_local_variable = {
			name = dmsd_gold
			multiply = {
				value = @dmsd_eff_baseline
				divide = { value = local_var:dmsd_gold min = @dmsd_eff_baseline }
				pow = @dmsd_eff_exponent
			}
		}

		dmsd_accumulate_culture = yes
	}
}
